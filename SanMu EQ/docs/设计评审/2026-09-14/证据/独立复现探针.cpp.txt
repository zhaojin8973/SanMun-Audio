#include "dsp/EqProcessor.h"
#include <cmath>
#include <cstdio>
#include <algorithm>
#include <memory>
constexpr double sr = 48000.0;
constexpr int bs = 512;
void setup(EqProcessor& eq, int mode, bool enabled=false, int placement=0, bool solo=false) {
    eq.prepare({sr, bs, 2});
    for(int i=0;i<Params::NUM_BANDS;++i)
        eq.updateBand(i,Params::Peak,i==0 && enabled,1000.0f,i==0?6.0f:0.0f,1.0f,i==0&&solo,0,placement,sr);
    eq.setProcessingMode(mode);
}
void tone(int mode,int placement,bool anti=false,bool disabledSolo=false) {
    auto eq=std::make_unique<EqProcessor>();
    setup(*eq,mode,!anti&&!disabledSolo,placement,disabledSolo);
    juce::AudioBuffer<float> b(2,bs);
    double sums[2]={0,0};int n=0;
    for(int block=0;block<140;++block) {
        for(int i=0;i<bs;++i) {
            float v=0.1f*std::sin(2*3.141592653589793*1000*(block*bs+i)/sr);
            b.setSample(0,i,v);b.setSample(1,i,anti?-v:v);
        }
        eq->process(b);
        if(block>=40) {for(int c=0;c<2;++c)for(int i=0;i<bs;++i)sums[c]+=b.getSample(c,i)*b.getSample(c,i);n+=bs;}
    }
    float peak=-1000;
    for(int i=0;i<EqProcessor::SPECTRUM_POINTS;++i)peak=std::max(peak,eq->getSpectrumPoint(i));
    std::printf("正弦: 模式=%d 路由=%d 反相=%d 禁用段独听=%d 左增益=%.3f dB 右增益=%.3f dB 左RMS=%.6f 频谱峰=%.2f dBFS\n",mode,placement,anti,disabledSolo,20*std::log10(std::sqrt(sums[0]/n)/0.070710678),20*std::log10(std::sqrt(sums[1]/n)/0.070710678),std::sqrt(sums[0]/n),peak);
}
void impulse(bool bypass) {
    auto eq=std::make_unique<EqProcessor>();setup(*eq,1);
    eq->setGlobalBypass(bypass);
    juce::AudioBuffer<float> b(2,bs);
    for(int i=0;i<30;++i) {b.clear();eq->process(b);}
    float max=0;int pos=-1;
    for(int block=0;block<30;++block) {
        b.clear();if(block==0){b.setSample(0,0,1);b.setSample(1,0,1);}eq->process(b);
        for(int i=0;i<bs;++i)if(std::abs(b.getSample(0,i))>max){max=std::abs(b.getSample(0,i));pos=block*bs+i;}
    }
    std::printf("脉冲: 线性相位 旁通=%d 上报延迟=%d 实际峰位置=%d 峰值=%.6f\n",bypass,eq->getLatencySamples(),pos,max);
}
void calibration() {
    auto eq=std::make_unique<EqProcessor>();setup(*eq,0);
    juce::AudioBuffer<float> b(2,bs);
    for(int block=0;block<140;++block) {
        for(int i=0;i<bs;++i) {
            float v=0.1f*std::sin(2*3.141592653589793*1125*(block*bs+i)/sr);
            b.setSample(0,i,v);b.setSample(1,i,v);
        }
        eq->process(b);
    }
    float peak=-1000;
    for(int i=0;i<EqProcessor::SPECTRUM_POINTS;++i)peak=std::max(peak,eq->getSpectrumPoint(i));
    std::printf("频谱标定: 全部EQ关闭 1125Hz整bin正弦 输入峰值=-20.00dBFS 频谱峰=%.3fdBFS\n",peak);
}
int main() {
    tone(0,Params::PlacementLeft);tone(1,Params::PlacementLeft);
    tone(0,0,true);tone(0,0,false,true);
    impulse(false);impulse(true);calibration();
}
